```typescript
const lastNode = newFormula[newFormula.length - 1];
if (lastNode && lastNode.type === 'number') {
  // 防止多个小数点
  if (num === '.' && lastNode.value.includes('.')) {
    return;
  }
  // 追加到上一个数字
  const updated = [...newFormula];
  updated[updated.length - 1] = {
    ...lastNode,
    value: lastNode.value + num
  };
  setNewFormula(updated);
} else {
  // 创建新数字节点
  setNewFormula([...newFormula, { type: 'number', value: num }]);
}
```

### 3. 公式验证增强

**更新**: `validateFormula` 函数

**新增验证规则**:
```typescript
// 操作数类型(科目或数字)
const isOperand = (node: FormulaNode) => 
  node.type === 'account' || node.type === 'number';

// 1. 不允许连续的操作数(中间必须有运算符)
if (isOperand(current) && isOperand(next)) {
  return { valid: false, error: '两个操作数之间必须有运算符' };
}

// 2. 检查数字格式
if (current.type === 'number') {
  const num = parseFloat(current.value);
  if (isNaN(num)) {
    return { valid: false, error: `无效的数字: ${current.value}` };
  }
  // 不允许以小数点开头或结尾
  if (current.value.startsWith('.') || current.value.endsWith('.')) {
    return { valid: false, error: '数字不能以小数点开头或结尾' };
  }
}
```

**验证覆盖**:
- ✅ 括号匹配
- ✅ 运算符位置
- ✅ 操作数连续性(科目、数字都视为操作数)
- ✅ 数字格式有效性
- ✅ 小数点位置检查

### 4. 公式计算升级

**更新**: `evaluateFormulaForMonth` 函数

**原有逻辑** (仅支持加减):
```typescript
let result = 0;
let currentOperator: '+' | '-' = '+';
formula.forEach(node => {
  if (node.type === 'operator') {
    currentOperator = node.value as '+' | '-';
  } else if (node.type === 'account') {
    const value = calculateMonthlyActual(account, month);
    result = currentOperator === '+' ? result + value : result - value;
  }
});
```

**新逻辑** (支持完整运算符 + 数字):
```typescript
const evaluateFormulaForMonth = (formula: FormulaNode[], month: number): number => {
  try {
    // 将公式节点转换为表达式字符串
    const expression = formula.map(node => {
      if (node.type === 'account') {
        const account = accounts.find(a => a.id === node.value);
        if (!account) return '0';
        return calculateMonthlyActual(account, month).toString();
      } else if (node.type === 'number') {
        return node.value;
      } else if (node.type === 'operator') {
        // 将乘除符号转换为标准运算符
        if (node.value === '×') return '*';
        if (node.value === '÷') return '/';
        return node.value;
      } else {
        return node.value; // parenthesis
      }
    }).join(' ');
    
    // 使用 Function 构造函数安全地计算表达式
    const result = new Function('return ' + expression)();
    return isFinite(result) ? result : 0;
  } catch (error) {
    console.error('公式计算错误:', error);
    return 0;
  }
};
```

**改进点**:
- ✅ 支持完整运算符: `+` `-` `×`(乘) `÷`(除)
- ✅ 支持括号运算优先级
- ✅ 支持数字直接参与计算
- ✅ 自动符号转换(`×` → `*`, `÷` → `/`)
- ✅ 安全的表达式求值(使用 Function 构造函数)
- ✅ 错误处理(返回0避免崩溃)

## ⚠️ 注意事项

1. **数字精度**
   - JavaScript 浮点数精度限制
   - 建议税率等系数不超过4位小数

2. **除零保护**
   - 计算时会检查结果是否有限(`isFinite`)
   - 无效结果返回0

3. **数字输入规范**
   - 必须是有效的数字格式
   - 不能以小数点开头或结尾
   - 例: `0.13` ✅  `.13` ❌  `13.` ❌
