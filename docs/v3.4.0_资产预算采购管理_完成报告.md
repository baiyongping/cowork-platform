```javascript
// 剩余预算检查
async function checkRemainingBudget(projectId, newBatchBudget) {
  const project = await db.collection('assetBudgetProjects')
    .doc(projectId).get();
  
  const remainingBudget = project.data.budgetAmount - project.data.allocatedAmount;
  
  if (newBatchBudget > remainingBudget) {
    throw new Error(`批次预算超过剩余预算`);
  }
  
  return true;
}

// 自动生成任务
async function createBatchTasks(batchData) {
  const tasks = [
    {
      name: `${batchData.batchName}-申请`,
      type: '采购任务',
      purchasePhase: '采购申请',
      canStart: true
    },
    {
      name: `${batchData.batchName}-执行`,
      type: '采购任务',
      purchasePhase: '采购执行',
      canStart: false
    },
    {
      name: `${batchData.batchName}-验收`,
      type: '采购任务',
      purchasePhase: '采购验收',
      canStart: false
    }
  ];
  
  await db.collection('tasks').add(tasks);
}
```

```javascript
// 任务完成自动解锁下一阶段
async function onTaskComplete(taskId) {
  const task = await db.collection('tasks').doc(taskId).get();
  
  // 查找下一阶段任务
  const nextPhase = getNextPhase(task.data.purchasePhase);
  if (!nextPhase) return;
  
  // 解锁下一阶段任务
  await db.collection('tasks').where({
    purchaseBatchId: task.data.purchaseBatchId,
    purchasePhase: nextPhase
  }).update({
    canStart: true
  });
}

function getNextPhase(currentPhase) {
  const phases = ['采购申请', '采购执行', '采购验收'];
  const index = phases.indexOf(currentPhase);
  return index < phases.length - 1 ? phases[index + 1] : null;
}
```

### 1. 项目化管理
- 每个采购预算都是一个独立项目
- 完整的项目生命周期管理

### 2. 多批次采购

```
采购申请 → 采购执行 → 采购验收
```

- 实时计算剩余预算

- 自动预算分配和回收

## 📊 技术亮点

### 1. 数据一致性保证
```javascript
// 使用事务保证数据一致性
async function createBatchWithTransaction(batchData) {
  const transaction = db.startTransaction();
  
  try {
    // 创建批次
    await transaction.collection('purchaseBatches').add(batchData);
    
    // 更新预算项目
    await transaction.collection('assetBudgetProjects')
      .doc(batchData.budgetProjectId)
      .update({
        allocatedAmount: _.inc(batchData.batchBudget)
      });
    
    // 创建任务
    await createBatchTasks(batchData);
    
    await transaction.commit();
  } catch (error) {
    await transaction.rollback();
    throw error;
  }
}
```

### 2. 智能依赖解锁

```javascript
// 任务完成触发下一阶段解锁
async function onTaskComplete(taskId) {

    // 解锁下一阶段任务
```

### 3. 实时统计汇总
```javascript
// 采购预算总览统计
async function calculateProjectStatistics(projectId) {
  const batches = await db.collection('purchaseBatches')
    .where({ budgetProjectId: projectId })
    .get();
  
  return {
    totalBatches: batches.data.length,
    completedBatches: batches.data.filter(b => b.status === '已完成').length,
    allocatedAmount: batches.data.reduce((sum, b) => sum + b.batchBudget, 0),
    actualAmount: batches.data.reduce((sum, b) => sum + b.batchActual, 0)
  };
}
```
