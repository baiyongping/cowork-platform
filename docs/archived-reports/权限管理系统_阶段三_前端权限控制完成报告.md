## 🎯 权限控制流程

```
用户操作
  ↓
前端权限检查（UI层）
  ├─ 按钮显示/隐藏 ← PermissionButton
  ├─ 路由访问控制 ← PermissionRoute
  ├─ 菜单权限过滤 ← filterMenus()
  └─ 数据列表过滤 ← filterData()
  ↓
发送API请求
  ↓
后端权限检查（业务层）
  ├─ JWT认证 ← authenticate
  ├─ 功能权限 ← checkFunction
  └─ 数据权限 ← checkEdit/checkDelete
  ↓
业务逻辑处理
  ↓
返回响应
```

## 🎨 使用示例

### 示例1: 任务列表页面

```tsx
import React, { useState, useEffect } from 'react';
import { PermissionButton } from '@/components/PermissionButton';
import { usePermissionContext } from '@/contexts/PermissionContext';

function TaskListPage() {
  const [tasks, setTasks] = useState([]);
  const { filterData, attachPermissions } = usePermissionContext();
  const currentUser = useCurrentUser();

  useEffect(() => {
    loadTasks();
  }, []);

  const loadTasks = async () => {
    const allTasks = await fetchTasks();
    
    // ⭐ 过滤用户可访问的任务
    const accessibleTasks = filterData(allTasks);
    
    // ⭐ 附加权限标识
    const tasksWithPerms = attachPermissions(accessibleTasks);
    
    setTasks(tasksWithPerms);
  };

  return (
    <div>
      {/* ⭐ 创建按钮权限控制 */}
      <PermissionButton
        currentUser={currentUser}
        module="tasks"
        action="create"
        onClick={handleCreate}
      >
        新建任务
      </PermissionButton>

      <table>
        <tbody>
          {tasks.map(task => (
            <tr key={task._id}>
              <td>{task.name}</td>
              <td>
                {/* ⭐ 编辑按钮权限控制 */}
                <PermissionButton
                  currentUser={currentUser}
                  module="tasks"
                  action="edit"
                  dataItem={task}
                  onClick={() => handleEdit(task)}
                >
                  编辑
                </PermissionButton>
                
                {/* ⭐ 删除按钮权限控制 */}
                <PermissionButton
                  currentUser={currentUser}
                  module="tasks"
                  action="delete"
                  dataItem={task}
                  onClick={() => handleDelete(task)}
                >
                  删除
                </PermissionButton>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}
```

### 示例3: 菜单权限过滤

```tsx
import { usePermissionContext } from '@/contexts/PermissionContext';

function Sidebar() {
  const { filterMenus } = usePermissionContext();

  const allMenus = [
    { key: 'tasks', label: '任务管理', path: '/tasks' },
    { key: 'opportunities', label: '商机管理', path: '/opportunities' },
    { key: 'projects', label: '项目管理', path: '/projects' },
    { 
      key: 'system', 
      label: '系统设置', 
      path: '/system',
      children: [
        { key: 'roles', label: '角色管理', path: '/system/roles' },
        { key: 'types', label: '类型管理', path: '/system/types' }
      ]
    }
  ];

  // ⭐ 根据用户权限过滤菜单
  const accessibleMenus = filterMenus(allMenus);

  return (
    <nav>
      {accessibleMenus.map(menu => (
        <MenuItem key={menu.key} menu={menu} />
      ))}
    </nav>
  );
}
```

```typescript

checkViewPermission(dataItem: DataItem): boolean
checkEditPermission(dataItem: DataItem): boolean
checkDeletePermission(dataItem: DataItem): boolean

// 路由权限
checkRoutePermission(routePath: string): boolean

// 按钮权限
checkButtonPermission(module, action, dataItem?): boolean
```

### 数据过滤函数

```typescript
// 过滤可访问数据
filterData<T>(dataList: T[]): T[]

// 附加权限标识
attachPermissions<T>(dataList: T[]): Array<T & { canEdit, canDelete }>
```
