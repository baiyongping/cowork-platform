### 方案二:使用云函数批量创建

创建一个临时云函数:

```javascript
// cloudfunctions/createCollections/index.js
const tcb = require('@cloudbase/node-sdk');

exports.main = async (event, context) => {
  const app = tcb.init({
    env: 'cowork-9gg9oocb516be5fb'
  });
  
  const db = app.database();
  
  const collections = [
    'users',
    'departments',
    'tasks',
    'opportunities',
    'projects',
    'goals',
    'quarterly_measures',
    'issues',
    'issue_replies',
    'messages',
    'task_comments',
    'type_settings',
    'role_permissions',
    'operation_logs',
    'annual_budgets',
    'budget_accounts',
    'budget_execution',
    'product_order_forecast',
    'purchaseBatches',
    'sales_goals',
    'login_codes',
    'sms_codes'
  ];
  
  const results = [];
  
  for (const collectionName of collections) {
    try {
      await db.createCollection(collectionName);
      results.push({ collection: collectionName, status: 'success' });
    } catch (error) {
      results.push({ 
        collection: collectionName, 
        status: 'error',
        message: error.message 
      });
    }
  }
  
  return {
    success: true,
    results
  };
};
```

**部署并调用**:
```bash

# 调用云函数

```

### 方案三:迁移基础数据

对于有数据的集合,需要迁移:

1. **departments** (4 条数据)
2. **goals** (1 条数据)
3. **quarterly_measures** (1 条数据)
4. **type_settings** (3 条数据)
5. **role_permissions** (1 条数据)

**迁移脚本**:
```javascript
// cloudfunctions/migrateData/index.js
const tcb = require('@cloudbase/node-sdk');

exports.main = async (event, context) => {
  const sourceEnv = 'jihua-oa-dev-3goht9irae4d949f';
  const targetEnv = 'cowork-9gg9oocb516be5fb';
  
  const sourceApp = tcb.init({ env: sourceEnv });
  const targetApp = tcb.init({ env: targetEnv });
  
  const collectionsToMigrate = [
    'departments',
    'goals',
    'quarterly_measures',
    'type_settings',
    'role_permissions'
  ];
  
  const results = [];
  
  for (const collectionName of collectionsToMigrate) {
    try {
      // 从测试环境读取数据
      const sourceData = await sourceApp
        .database()
        .collection(collectionName)
        .get();
      
      // 写入生产环境
      if (sourceData.data && sourceData.data.length > 0) {
        for (const item of sourceData.data) {
          // 删除 _id 字段,让生产环境自动生成
          delete item._id;
          
          await targetApp
            .database()
            .collection(collectionName)
            .add(item);
        }
        
        results.push({
          collection: collectionName,
          status: 'success',
          count: sourceData.data.length
        });
      } else {
        results.push({
          collection: collectionName,
          status: 'skipped',
          reason: 'no data'
        });
      }
    } catch (error) {
      results.push({
        collection: collectionName,
        status: 'error',
        message: error.message
      });
    }
  }
  
  return {
    success: true,
    results
  };
};
```

## ⚠️ 注意事项

1. **权限配置**:
   - 迁移完成后需要配置集合权限
   - 建议使用安全规则

2. **索引优化**:
   - 创建必要的索引提升性能
   - 唯一索引避免数据重复

3. **数据验证**:
   - 迁移后验证数据完整性
   - 检查关联数据是否正确

4. **回滚方案**:
   - 迁移前备份测试环境数据
   - 保留原始数据以便回滚
